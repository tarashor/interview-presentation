% Lesson 2
\begin{frame}{How to Use This Course Effectively}
  \begin{itemize}
    \item \textbf{Course Approach:} We will start by showing the brute-force solution, and then transition to the optimal solution.
    \item \textbf{In an Interview (Ideal Case):}
      \begin{enumerate}
        \item \textbf{Identify:} Recognize pattern triggers in the problem text
        \item \textbf{Trace:} Dry-run the logic on a small input
        \item \textbf{Implement:} Code the solution without looking at hints
      \end{enumerate}
  \end{itemize}
\end{frame}

% Lesson 3
\begin{frame}{Interview Mindset: Thinking in Patterns}
  \begin{itemize}
    \item Stop looking at the "Story" (e.g., "Alice has 3 apples...")
    \item Start looking at the \textbf{Data Flow}
  \end{itemize}
  
  \begin{table}[]
    \scriptsize
    \begin{tabular}{ll}
        \toprule
        \textbf{Problem Trait} & \textbf{Pattern Strategy} \\ \midrule

        Searching a sorted range & Binary Search \\
        Shortest path in unweighted graph & BFS \\ \bottomrule
    \end{tabular}
  \end{table}
\end{frame}

% Lesson 5
\begin{frame}{Common Interview Mistakes}
  \begin{itemize}
    \item \textbf{Mistake 1:} Starting to code before the logic is 100\% clear \only<2->{\emoji{cross-mark}}
    \item \textbf{Mistake 2:} Forgetting to state Big O complexity \only<3->{\emoji{cross-mark}}
    \item \textbf{Mistake 3:} Not testing for $n=0$ or $n=1$ \only<4->{\emoji{cross-mark}}
  \end{itemize}
  \vspace{0.3cm}
  \textbf{If Stuck:} Start with a brute-force solution, optimize what you can, and carefully listen to your interviewer. \\
  \vspace{0.3cm}
  \textbf{Correction:} Think out loud. Your interviewer is your collaborator, not your judge.
\end{frame}
