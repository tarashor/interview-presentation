% Slide 1: Concept
\begin{frame}{What is the Merge Intervals Pattern?}
  \begin{itemize}
    \item \textbf{The Core Idea:} Process overlapping intervals by sorting them and merging adjacent ranges.
    \item \textbf{Why it works:} Once sorted by start time, overlapping intervals will be adjacent. We can merge them in a single linear pass.
  \end{itemize}
\end{frame}

\begin{frame}{Merging Logic}
  Given two sorted intervals $A$ and $B$, they overlap if:
  $$B.\text{start} \le A.\text{end}$$
  If they overlap, we merge them into a single interval $C$ where:
  $$C.\text{start} = A.\text{start}$$
  $$C.\text{end} = \max(A.\text{end}, B.\text{end})$$
  Otherwise, they do not overlap, and we add $A$ to our output list and set $B$ as the new active interval.
\end{frame}

% Problem 1: Merge Intervals
\begin{frame}[fragile]{Problem: Consolidating Overlapping Intervals}
  \textbf{Scenario:}
  Given an array of intervals [start, end], merge all overlapping intervals and return a list of non-overlapping intervals covering the same range.
  
  \vspace{0.3cm}
  \textbf{Examples:}
  \begin{itemize}
    \item \textbf{Input:} \texttt{[[1,3],[2,6],[8,10],[15,18]]} $\rightarrow$ \textbf{Output:} \texttt{[[1,6],[8,10],[15,18]]}
  \end{itemize}
\end{frame}

% Problem 2: Insert Interval
\begin{frame}[fragile]{Problem: Inserting into Non-Overlapping Intervals}
  \textbf{Scenario:}
  Given a set of non-overlapping intervals sorted by start time, insert a new interval and merge if it overlaps with existing intervals.
  
  \vspace{0.3cm}
  \textbf{Examples:}
  \begin{itemize}
    \item \textbf{Input:} \texttt{intervals = [[1,3],[6,9]], newInterval = [2,5]} $\rightarrow$ \textbf{Output:} \texttt{[[1,5],[6,9]]}
  \end{itemize}
\end{frame}

% Problem 3: Employee Free Time
\begin{frame}[fragile]{Problem: Common Available Time Slots}
  \textbf{Scenario:}
  Given schedules of multiple users containing busy intervals, find the free time intervals common to all users.
  
  \vspace{0.3cm}
  \textbf{Examples:}
  \begin{itemize}
    \item \textbf{Input:} \texttt{[[[1,2],[5,6]],[[1,3]],[[4,10]]]} $\rightarrow$ \textbf{Output:} \texttt{[[3,4]]}
  \end{itemize}
\end{frame}
