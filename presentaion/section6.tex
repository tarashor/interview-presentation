% Slide 1: Concept
\begin{frame}{What is the Prefix Sum Pattern?}
  \begin{itemize}
    \item \textbf{The Core Idea:} Precompute cumulative sums of an array: `prefix[i] = nums[0] + ... + nums[i]`.
    \item \textbf{Why it works:} Allows you to calculate the sum of any subarray `nums[i...j]` in $O(1)$ time:
      $$\text{Sum}(i, j) = \text{prefix}[j] - \text{prefix}[i-1]$$
  \end{itemize}
\end{frame}

\begin{frame}{Combining Prefix Sum with HashMap}
  To find subarrays summing to $K$ in $O(N)$ time:
  \begin{itemize}
    \item Maintain running cumulative sum `currentSum`.
    \item At each step, we look for a previous prefix sum that equals `currentSum - K`.
    \item Use a HashMap to store frequencies of prefix sums seen so far.
    \item \textbf{Key Modulo Trick:} For divisibility by $K$, store `currentSum % K` (handling negative mod values properly).
  \end{itemize}
\end{frame}

% Problem 1: Subarray Sum Equals K
\begin{frame}[fragile]{Problem: Subarrays Summing to Target}
  \textbf{Scenario:}
  Given an array of integers `array` and a target value `target`, return the total count of contiguous subarrays whose elements sum up exactly to `target`.
  
  \vspace{0.3cm}
  \textbf{Examples:}
  \begin{itemize}
    \item \textbf{Input:} \texttt{array = [1, 1, 1], target = 2} $\rightarrow$ \textbf{Output:} \texttt{2}
  \end{itemize}
\end{frame}

% Problem 2: Subarray Sums Divisible by K
\begin{frame}[fragile]{Problem: Subarrays Sum Divisible by K}
  \textbf{Scenario:}
  Given an array of integers `array` and a value `divisor`, return the number of non-empty contiguous subarrays that have a sum divisible by `divisor`.
  
  \vspace{0.3cm}
  \textbf{Examples:}
  \begin{itemize}
    \item \textbf{Input:} \texttt{array = [4, 5, 0, -2, -3, 1], divisor = 5} $\rightarrow$ \textbf{Output:} \texttt{7}
  \end{itemize}
\end{frame}
