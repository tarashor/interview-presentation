% Slide 1: Concept
\begin{frame}{What is the Sliding Window Pattern?}
  \begin{itemize}
    \item \textbf{The Core Idea:} Maintain a contiguous window over a linear data structure (array/string) that dynamically expands or contracts.
    \item \textbf{Why it works:} Converts nested $O(n^2)$ loops searching subarrays into a single $O(n)$ pass by reusing computed information.
    \item \textbf{Common Triggers:}
      \begin{itemize}
        \item Subarray or substring problems.
        \item Finding shortest, longest, or target length matching constraint.
        \item Tracking unique elements/frequencies in a range.
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Dynamic vs. Fixed Window}
  \begin{itemize}
    \item \textbf{Fixed Window:} Size $K$ is constant. Move window by adding next element and removing the leftmost.
    \item \textbf{Dynamic Window:} Size changes.
      \begin{itemize}
        \item Expand the window by moving `right` pointer.
        \item If constraint is violated, contract window by moving `left` pointer until constraint is satisfied again.
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Longest Substring Without Repeating Characters}
    \textbf{Problem:} Find length of the longest substring with unique characters.

    \begin{columns}
        \begin{column}{0.45\textwidth}
            \begin{itemize}
                \item<1-> \textbf{Expand:} Move \texttt{right} pointer.
                \item<2-> \textbf{Detect:} If \texttt{s[right]} is in Map...
                \item<3-> \textbf{Jump:} Move \texttt{left} to \texttt{Map[char] + 1}.
                \item<4-> \textbf{Update:} Update Max Length.
            \end{itemize}

            \begin{onlyenv}<5->
            \begin{lstlisting}[language=Kotlin]
for (right in s.indices) {
    if (map.containsKey(s[right])) {
        left = maxOf(left, map[s[right]]!! + 1)
    }
    map[s[right]] = right
    maxL = maxOf(maxL, right - left + 1)
}
            \end{lstlisting}
            \end{onlyenv}
        \end{column}

        \begin{column}{0.55\textwidth}
            \centering
            \textbf{Step-by-Step Window}
            \vspace{0.5cm}

            \begin{tikzpicture}[scale=0.8]
                % String: a b c a
                \foreach \x/\val in {0/a, 1/b, 2/c, 3/a}
                    \node[draw, minimum size=0.8cm] (c\x) at (\x,0) {\val};

                \draw<1-2>[->, blue, thick] (0,-0.8) -- (0,-0.4) node[below=0.4] {L};

                \draw<2>[->, red, thick] (2, 0.8) -- (2, 0.4) node[above=0.4] {R};

                % State 2: Conflict at R=3

                % State 3: The Jump
                \draw<4->[->, blue, thick] (1,-0.8) -- (1,-0.4) node[below=0.4] {L};
                \draw<4->[->, red, thick] (3, 0.8) -- (3, 0.4) node[above=0.4] {R};
                \draw<4->[green, thick, rounded corners] (0.6, -0.5) rectangle (3.4, 0.5);
            \end{tikzpicture}

    \textbf{Problem:} Does $s_2$ contain a permutation of $s_1$?

    \begin{center}
    \begin{tikzpicture}

        \draw<3>[green, ultra thick] (2.6,-0.4) rectangle (4.4,0.4);
        \node<3>[green] at (3.5, -0.8) {Permutation Found!};
    \end{tikzpicture}
    \end{center}

    \begin{block}{HashMap Slide Logic}<4->
\begin{lstlisting}[language=Kotlin]
// 1. Add right character
s2Map[s2[right]] = s2Map.getOrDefault(s2[right], 0) + 1
// 2. Remove left character
if (s2Map[s2[left]] == 1) s2Map.remove(s2[left])
else s2Map[s2[left]] = s2Map[s2[left]]!! - 1
// 3. Compare Maps
if (s1Map == s2Map) return true
\end{lstlisting}
    \end{block}
\end{frame}
