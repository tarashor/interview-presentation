private fun atMostK(nums: IntArray, k: Int): Int {
    var L = 0, count = 0, res = 0
    for (R in nums.indices) {
        if (map.getOrDefault(nums[R], 0) == 0) k--
        map[nums[R]] = map.getOrDefault(nums[R], 0) + 1
        
        while (k < 0) {
            map[nums[L]] = map[nums[L]]!! - 1
            if (map[nums[L]] == 0) k++
            L++
        }
        res += (R - L + 1) // Counts all valid ending at R
    }
    return res
}
    \end{lstlisting}
    \end{onlyenv}
