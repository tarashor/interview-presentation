% Lesson 2
\begin{frame}{How to Use This Course Effectively}
  \begin{itemize}
    \item \textbf{The Three-Step Loop:}
      \begin{enumerate}
        \item \textbf{Identify:} Recognize pattern triggers in the problem text \only<2->{\emoji{magnifying-glass-tilted-right}}
        \item \textbf{Trace:} Dry-run the logic on a small input \only<3->{\emoji{pencil}}
        \item \textbf{Implement:} Code the solution without looking at hints \only<4->{\emoji{keyboard}}
      \end{enumerate}
    \item Use the provided Cheat Sheets for last-minute review.
  \end{itemize}
\end{frame}

    \item Start looking at the \textbf{Data Flow}
  \end{itemize}

\begin{frame}{Common Interview Mistakes}
  \begin{itemize}
    \item \textbf{Mistake 1:} Starting to code before the logic is 100\% clear \only<2->{\emoji{cross-mark}}
    \item \textbf{Mistake 2:} Forgetting to state Big O complexity \only<3->{\emoji{cross-mark}}
    \item \textbf{Mistake 3:} Not testing for $n=0$ or $n=1$ \only<4->{\emoji{cross-mark}}
  \end{itemize}
  \vspace{0.5cm}
  \textbf{Correction:} Think out loud. Your interviewer is your collaborator, not your judge.
